  <button id="spinBtn" onclick="spin()">開始抽任務！</button>

  <div class="result-box" id="result">點擊按鈕，抽取今天的減塑任務！</div>

  <script>
    // 20 個減塑任務清單
    const tasks = [
      "自備質感環保袋，堅決拒絕免費塑膠袋",
      "攜帶隨身環保水壺，從此不買瓶裝水",
      "包包常備個人餐具，拒用一次性免洗筷",
      "攜帶環保杯買手搖飲，每杯現折 5 元！",
      "拒絕塑膠吸管，直接對口喝或用不鏽鋼管",
      "採買傳統市場蔬果，不拿二次分裝透明袋",
      "檢視首週垃圾桶，記錄自己少製造了多少垃圾",
      "選擇「裸賣蔬果」，拒買過度保鮮膜包裝",
      "改用固體肥皂或洗髮餅，告別塑膠瓶罐",
      "外帶食物自備不鏽鋼餐盒或矽膠密封袋",
      "拒絕購買過度精美包裝的禮盒商品",
      "食材分裝改用玻璃罐，取代保鮮膜消耗",
      "減少衝動網購，降低大量紙箱與塑膠包材",
      "分享你的減塑社群動態，擴大綠色影響力！",
      "朝聖無包裝商店，自備容器秤重買雜貨",
      "拒收不必要的小紙條、實體發票與明細",
      "舊衣物或廢棄布料動手改造成環保提袋",
      "用實際購買行動，支持綠色環保友善店家",
      "清理家中一次性塑膠，並做好極致分類",
      "傾囊相授！傳授親友一個超實用減塑妙招"
    ];

    const colors = ["#2d6a4f", "#40916c", "#52b788", "#74c69d", "#95d5b2", "#b7e4c7"];
    const total = tasks.length;
    const sliceAngle = 360 / total;
    const svg = document.getElementById("wheel");
    const resultBox = document.getElementById("result");
    const spinBtn = document.getElementById("spinBtn");

    let currentRotation = 0;

    // 繪製 SVG 輪盤扇形與文字
    function drawWheel() {
      for (let i = 0; i < total; i++) {
        const startAngle = i * sliceAngle;
        const endAngle = (i + 1) * sliceAngle;
        
        // 計算扇形邊界座標 (半徑 250，圓心 250, 250)
        const x1 = 250 + 250 * Math.cos((Math.PI * (startAngle - 90)) / 180);
        const y1 = 250 + 250 * Math.sin((Math.PI * (startAngle - 90)) / 180);
        const x2 = 250 + 250 * Math.cos((Math.PI * (endAngle - 90)) / 180);
        const y2 = 250 + 250 * Math.sin((Math.PI * (endAngle - 90)) / 180);

        const pathData = `M 250 250 L ${x1} ${y1} A 250 250 0 0 1 ${x2} ${y2} Z`;
        const fillColor = colors[i % colors.length];

        // 創建扇形
        const path = document.createElementNS("http://www.w3.org/2000/svg", "path");
        path.setAttribute("d", pathData);
        path.setAttribute("fill", fillColor);
        path.setAttribute("stroke", "#ffffff");
        path.setAttribute("stroke-width", "2");
        svg.appendChild(path);

        // 創建文字標籤 (縮簡顯示前 6 個字)
        const textAngle = startAngle + sliceAngle / 2;
        const text = document.createElementNS("http://www.w3.org/2000/svg", "text");
        const textRadius = 170;
        const tx = 250 + textRadius * Math.cos((Math.PI * (textAngle - 90)) / 180);
        const ty = 250 + textRadius * Math.sin((Math.PI * (textAngle - 90)) / 180);

        text.setAttribute("x", tx);
        text.setAttribute("y", ty);
        text.setAttribute("fill", "#ffffff");
        text.setAttribute("font-size", "12");
        text.setAttribute("font-weight", "bold");
        text.setAttribute("text-anchor", "middle");
        text.setAttribute("dominant-baseline", "middle");
        text.setAttribute("transform", `rotate(${textAngle}, ${tx}, ${ty})`);
        
        const label = tasks[i].length > 6 ? tasks[i].substring(0, 6) + ".." : tasks[i];
        text.textContent = label;
        svg.appendChild(text);
      }
    }

    // 旋轉控制邏輯
    function spin() {
      spinBtn.disabled = true;
      resultBox.textContent = "抽籤中...";

      // 設定隨機旋轉圈數 (5 至 10 圈)
      const extraRounds = Math.floor(Math.random() * 5) + 5;
      const randomDegree = Math.floor(Math.random() * 360);
      const totalDegrees = extraRounds * 360 + randomDegree;

      currentRotation += totalDegrees;
      svg.style.transform = `rotate(${currentRotation}deg)`;

      // 動畫結束後計算結果 (4000ms 對應 CSS 的 4s 轉速)
      setTimeout(() => {
        const actualDegree = currentRotation % 360;
        // 計算頂點指標落在第幾個分頁
        const winningIndex = Math.floor((360 - actualDegree) / sliceAngle) % total;
        
        resultBox.textContent = "🎯 " + tasks[winningIndex];
        spinBtn.disabled = false;
      }, 4000);
    }

    // 初始化輪盤
    drawWheel();
  </script>
